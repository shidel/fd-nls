  -nv, --no-verbose                 вимкнути докладність, не стаючи
                                    тихим.
  -i,  --input-file=FILE            завантажити URL, знайдені у FILE.
  -F,  --force-html                 вважати вхідний файл як HTML.
  -B,  --base=URL                   додати URL перед відносними
                                    посиланнями.

Завантаження:
  -t,  --tries=NUMBER               встановити кількість спроб на NUMBER
                                    (0 безлімітно).
       --retry-connrefused          повторювати навіть при відмові
                                    з'єднання.
  -O,  --output-document=FILE       записувати документи у FILE.
  -nc, --no-clobber                 пропустити завантаження, що
                                    перезаписали б існуючі файли.
  -c,  --continue                   продовжити завантаження часткового
                                    файлу.
       --progress=TYPE              вибрати тип індикатора прогресу.
  -N,  --timestamping               не завантажувати знову, якщо не
                                    новіший за локальний.
  -S,  --server-response            вивести відповідь сервера.
       --spider                     нічого не завантажувати.
  -T,  --timeout=SECONDS            встановити всі тайм-аути на SECONDS.
       --dns-timeout=SECS           встановити тайм-аут DNS на SECS.
       --connect-timeout=SECS       встановити тайм-аут з'єднання
                                    на SECS.
       --read-timeout=SECS          встановити тайм-аут читання на SECS.
  -w,  --wait=SECONDS               чекати SECONDS між завантаженнями.
       --waitretry=SECONDS          чекати 1..SECONDS між спробами.
       --random-wait                чекати 0...2*WAIT сек між
                                    завантаженнями.
       --no-proxy                   прямо вимкнути проксі.
  -Q,  --quota=NUMBER               встановити квоту завантаження
                                    на NUMBER.
       --bind-address=ADDRESS       прив'язати до ADDRESS (ім'я/IP)
                                    локально.
       --limit-rate=RATE            обмежити швидкість завантаження
                                    на RATE.
       --no-dns-cache               вимкнути кешування DNS-запитів.
       --restrict-file-names=OS     обмежити символи в іменах для OS.
       --ignore-case                ігнорувати регістр при пошуку.
  -4,  --inet4-only                 підключатися лише до IPv4-адрес.
  -6,  --inet6-only                 підключатися лише до IPv6-адрес.
       --prefer-family=FAMILY       підключатися спочатку до цієї сім'ї,
                                    однієї з IPv6, IPv4 або none.
       --user=USER                  встановити FTP/HTTP користувача на
                                    USER.
       --password=PASS              встановити FTP/HTTP пароль на PASS.

Каталоги:
  -nd, --no-directories             не створювати каталоги.
  -x,  --force-directories          примусово створювати каталоги.
  -nH, --no-host-directories        не створювати каталоги хоста.
       --protocol-directories       використовувати ім'я протоколу в
                                    каталогах.
  -P,  --directory-prefix=PREFIX    зберігати файли у PREFIX/...
       --cut-dirs=NUMBER            ігнорувати NUMBER віддалених
                                    каталогів.

HTTP опції:
       --http-user=USER             встановити HTTP користувача на USER.
       --http-password=PASS         встановити HTTP пароль на PASS.
       --no-cache                   заборонити кешовані дані сервера.
  -E,  --html-extension             зберегти HTML з розширенням `.html'.
       --ignore-length              ігнорувати поле заголовка
                                    `Content-Length'.
       --header=STRING              вставити STRING між заголовками.
       --max-redirect               макс. перенаправлень на сторінку.
       --proxy-user=USER            встановити USER як користувача проксі.
       --proxy-password=PASS        встановити PASS як пароль проксі.
       --referer=URL                включити заголовок `Referer: URL'.
       --save-headers               зберегти HTTP-заголовки у файл.
  -U,  --user-agent=AGENT           ідентифікувати як AGENT, а не Wget.
       --no-http-keep-alive         вимкнути HTTP keep-alive.
       --no-cookies                 не використовувати файли cookie.
       --load-cookies=FILE          завантажити cookie з FILE до сесії.
       --save-cookies=FILE          зберегти cookie у FILE після сесії.
       --keep-session-cookies       завантажити та зберегти сесійні
                                    cookie.
       --post-data=STRING           використати метод POST; надіслати
                                    STRING.
       --post-file=FILE             використати метод POST; надіслати
                                    вміст FILE.
       --content-disposition        поважати заголовок
                                    Content-Disposition при виборі
                                    локальних імен (ЕКСПЕРИМЕНТАЛЬНО).
       --auth-no-challenge          Надіслати Basic HTTP-автентифікацію
                                    без очікування виклику сервера.

HTTPS (SSL/TLS) опції:
       --secure-protocol=PR         вибрати безпечний протокол, один з
                                    auto, SSLv2, SSLv3 та TLSv1.
       --no-check-certificate       не перевіряти сертифікат сервера.
       --certificate=FILE           файл клієнтського сертифіката.
       --certificate-type=TYPE      тип клієнтського сертифіката, PEM
                                    або DER.
       --private-key=FILE           файл приватного ключа.
       --private-key-type=TYPE      тип приватного ключа, PEM або DER.
       --ca-certificate=FILE        файл з набором сертифікатів CA.
       --ca-directory=DIR           каталог зі списком хешів CA.
       --random-file=FILE           файл з випадковими даними для SSL
                                    PRNG.
       --egd-file=FILE              файл, що називає EGD-сокет.

FTP опції:
       --ftp-user=USER              встановити FTP користувача на USER.
       --ftp-password=PASS          встановити FTP пароль на PASS.
       --no-remove-listing          не видаляти файли `.listing'.
       --no-glob                    вимкнути FTP globbing імен файлів.
       --no-passive-ftp             вимкнути "пасивний" режим передачі.
       --retr-symlinks              при рекурсії отримувати зв'язані
                                    файли.
       --preserve-permissions       зберігати права доступу віддалених
                                    файлів.

Рекурсивне завантаження:
  -r,  --recursive                  вказати рекурсивне завантаження.
  -l,  --level=NUMBER               макс. глибина рекурсії (inf/0
                                    нескінченно).
       --delete-after               видалити файли локально після
                                    завантаження.
  -k,  --convert-links              робити посилання в HTML локальними.
  -K,  --backup-converted           перед конвертацією X зберегти як
                                    X.orig.
  -m,  --mirror                     скорочення для -N -r -l inf
                                    --no-remove-listing.
  -p,  --page-requisites            отримати всі зображення тощо для HTML.
       --strict-comments            увімкнути строгу (SGML) обробку HTML
                                    коментарів.

Рекурсивне прийняття/відхилення:
  -A,  --accept=LIST                список прийнятних розширень (через
                                    кому).
  -R,  --reject=LIST                список відхилених розширень (через
                                    кому).
  -D,  --domains=LIST               список прийнятних доменів (через
                                    кому).
       --exclude-domains=LIST       список відхилених доменів (через
                                    кому).
       --follow-ftp                 переходити за FTP-посиланнями з HTML.
       --follow-tags=LIST           список HTML-тегів для відстеження
                                    (через кому).
       --ignore-tags=LIST           список ігнорованих HTML-тегів
                                    (через кому).
  -H,  --span-hosts                 переходити до чужих хостів при
                                    рекурсії.
  -L,  --relative                   переходити лише за відносними
                                    посиланнями.
  -I,  --include-directories=LIST   список дозволених каталогів.
  -X,  --exclude-directories=LIST   список виключених каталогів.
  -np, --no-parent                  не підніматися до батьківського
                                    каталогу.
